索引，切片，跳取#

索引（[index]），切片（[首，尾]，包前不包后），跳取（[首，尾，steps]）

s = 'ABCDEFGHIJKLMN'
s1 = s[0]
print('s[0] = ' + s1)   #s[0] = A
print('s[3] = '+ s[3])  #s[3] = D
print('倒数第三个数为：' + s[-3])   #倒数第三个数为：L
print('最后一个数为：' + s[-1])     #最后一个数为：N

移除首尾空白/指定字符#

s = "   hello, world!   "
result = s.strip()
print(result)
# 输出: 'hello, world!'


s = "***hello, world!***"
result = s.strip("*")
print(result)
# 输出: 'hello, world!'
